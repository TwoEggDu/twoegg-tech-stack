cmake_minimum_required(VERSION 3.20)
project(CppFoundationsExamples LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
enable_testing()
add_executable(alias cxx-01/alias.cpp)
add_executable(observer cxx-02/observer.cpp)
add_executable(failure_protocol cxx-02/failure_protocol.cpp)
add_executable(member_order cxx-03/member_order.cpp)
add_executable(handle cxx-10/handle/Handle.cpp cxx-10/handle/main.cpp)
foreach(target alias observer failure_protocol member_order handle)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -O0 -Wall -Wextra -Wpedantic -UNDEBUG)
    elseif(MSVC)
        target_compile_options(${target} PRIVATE /UNDEBUG)
    endif()
    add_test(NAME ${target} COMMAND ${target})
endforeach()
# fail/*.cpp are intentionally excluded from the default build.
